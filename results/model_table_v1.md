def check(name: str, got, want, ok: bool) -> None:
    CHECKS.append((name, str(got), str(want), bool(ok)))


def run_record(d: str) -> dict:
    """The completed training run of a stage directory (the one with the most steps among state == done)."""
    recs = [json.load(open(f)) for f in sorted(glob.glob(os.path.join(d, "run_*.json")))]
    done = [r for r in recs if r.get("state") == "done"]
    if not done:
        raise SystemExit(f"{d}: no completed run record")
    r = max(done, key=lambda r: r["global_step"])
    r["n_records"] = len(recs)
    return r


def graph_stats(run: str, prim: pd.DataFrame, events: np.ndarray, sub: str = "eval_val") -> dict:
    pe = pd.read_parquet(os.path.join(run, sub, "per_event.parquet"), columns=["event_id", "n_edges"])
    g = pe.groupby("event_id").n_edges.agg(["min", "max"])
    if not (g["min"] == g["max"]).all():
        raise SystemExit(f"{run}/{sub}: n_edges differs between cuts of one event")
    if not np.array_equal(np.sort(g.index.to_numpy(np.int64)), events):
        raise SystemExit(f"{run}/{sub}: event set differs from the validation set")
    ne = g["min"]
    code = ne.index.to_numpy(np.int64) // 100_000_000
    ce = pd.read_parquet(os.path.join(run, sub, "ceiling_particle.parquet"),
                         columns=["event_id", "particle_id", "dm_ceiling"])
    if ce.duplicated(["event_id", "particle_id"]).any():
        raise SystemExit(f"{run}/{sub}: duplicate particles in the ceiling table")
    m = prim.merge(ce, on=["event_id", "particle_id"], how="left")
    if m.dm_ceiling.isna().any():
        raise SystemExit(f"{run}/{sub}: primary particles without a ceiling entry")
    return {"pairs_ttbar": float(ne[code == TTBAR_CODE].mean()), "n_ttbar_events": int((code == TTBAR_CODE).sum()),
            "pairs_hnl": float(ne[code != TTBAR_CODE].mean()), "n_hnl_events": int((code != TTBAR_CODE).sum()),
            "k": int(m.dm_ceiling.astype(bool).sum()), "n": int(len(m)),
            "_edges": ne.sort_index().to_numpy(), "_ceil": m.sort_values(["event_id", "particle_id"]).dm_ceiling
            .astype(bool).to_numpy()}


def fmt(field: str, s: dict) -> str:
    if field == "pairs_int":
        return f"{s['pairs_ttbar']:,.0f}"
    if field == "pairs_k":
        return f"{s['pairs_ttbar'] / 1000:.1f}k"
    if field == "ceil3":
        return f"{s['k'] / s['n']:.3f}"
    if field == "ceil4":
        return f"{s['k'] / s['n']:.4f}"
    raise ValueError(field)


def main(argv=None) -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--out", default="results/model_table_v1.md")
    a = ap.parse_args(argv)
    from hnl.eval import bootstrap_tables as BT

    first = os.path.join(RUNS, MODELS[0][1].format(s=0), "eval_val", "per_event.parquet")
    events = np.sort(pd.read_parquet(first, columns=["event_id"]).event_id.unique().astype(np.int64))
    info = BT.info_table(events)
    prim = info[BT.masks(info)["eff_primary"]][["event_id", "particle_id"]].reset_index(drop=True)
    prim = prim.sort_values(["event_id", "particle_id"]).reset_index(drop=True)
    check("validation events", len(events), 2999, len(events) == 2999)
    check("primary particles per seed", len(prim), 1331, len(prim) == 1331)

    rows, stats = [], {}
    for label, pat, seeds, mlpat, rep in MODELS:
        for s in seeds:
            run = os.path.join(RUNS, pat.format(s=s))
            gnn = run_record(os.path.join(run, "gnn"))
            ml = run_record(os.path.join(RUNS, mlpat.format(s=s), "ml"))
            st = graph_stats(run, prim, events)
            stats[(label, s)] = st
            rows.append({"model": label, "seed": s, "run": pat.format(s=s), "train_graphs": gnn["data_split"][0],
                         "valmon_graphs": gnn["data_split"][1], "epochs": gnn["epochs_completed"],
                         "steps": gnn["global_step"], "parameters": gnn["n_parameters"],
                         "gnn_records": gnn["n_records"], "ml_epochs": ml["epochs_completed"],
                         "ml_steps": ml["global_step"], "pairs_ttbar": st["pairs_ttbar"],
                         "pairs_hnl": st["pairs_hnl"], "ceil_k": st["k"], "ceil_n": st["n"]})
            # run facts
            if label.startswith("DeZoort"):
                ok = gnn["global_step"] == 60000 and gnn["epochs_completed"] == 15 and gnn["data_split"][0] == 7999
                check(f"{label} s{s}: screening budget (15 epochs, 60000 steps, 7999 ttbar graphs)",
                      (gnn["epochs_completed"], gnn["global_step"], gnn["data_split"][0]), (15, 60000, 7999), ok)
            else:
                ok = gnn["global_step"] == 30 * gnn["data_split"][0] and gnn["epochs_completed"] == 30
                check(f"{label} s{s}: 30 epochs of one graph per step", gnn["global_step"],
                      f"30 x {gnn['data_split'][0]}", ok)
            key = "acorn" if label.startswith("acorn") else "Canto" if label.startswith("Canto") else label
            q, f, ln = PARAMS[key]
            line = open(f).read().split("\n")[ln - 1]
            check(f"{label} s{s}: parameters = {f}:{ln}", f"{gnn['n_parameters']:,}", q,
                  f"{gnn['n_parameters']:,}" == q and q in line)
            # the stored graph report of the same graph
            if rep is not None and os.path.exists(os.path.join(RUNS, rep[0].format(s=s))):
                path, which = rep
                path = os.path.join(RUNS, path.format(s=s))
                r = pd.read_csv(path).set_index("source").loc[which]
                ok = abs(r.pairs_ttbar - st["pairs_ttbar"]) < 1e-6 and abs(r.ceil_primary - st["k"] / st["n"]) < 1e-12 \
                    and int(r.n_primary) == st["n"]
                check(f"{label} s{s}: = graph report {path.replace(RUNS + '/', '')} ({which})",
                      f"{st['pairs_ttbar']:.3f} / {st['k'] / st['n']:.6f}", f"{r.pairs_ttbar:.3f} / {r.ceil_primary:.6f}",
                      ok)
            # acorn runs: the scored graphs equal the unscored metric-learning graphs (ceiling_ml), where stored
            cm = os.path.join(run, "ceiling_ml")
            if label.startswith("acorn") and os.path.exists(os.path.join(cm, "ceiling_particle.parquet")):
                st2 = graph_stats(run, prim, events, "ceiling_ml")
                ok = np.array_equal(st["_edges"], st2["_edges"]) and np.array_equal(st["_ceil"], st2["_ceil"])
                check(f"{label} s{s}: eval_val graph = ceiling_ml graph (every event, every primary particle)",
                      f"{st['pairs_ttbar']:.3f} / {st['k']}", f"{st2['pairs_ttbar']:.3f} / {st2['k']}", ok)
    # Canto-B and DeZoort run on the acorn (ttbar) graph of the same seed
    for label in ("Canto-B", "DeZoort IN", "DeZoort IN-L"):
        for (lab, s), st in stats.items():
            if lab != label:
                continue
            ref = stats[("acorn (ttbar)", s)]
            ok = np.array_equal(st["_edges"], ref["_edges"]) and np.array_equal(st["_ceil"], ref["_ceil"])
            check(f"{label} s{s}: graph = acorn (ttbar) s{s} graph (n_edges of every event, dm_ceiling of every "
                  f"primary particle)", f"{st['pairs_ttbar']:.3f} / {st['k']}", f"{ref['pairs_ttbar']:.3f} / {ref['k']}", ok)
    m2 = pd.read_csv(M2).set_index("source")
    for src, key in M2_ROWS.items():

        check(f"{key[0]} s{key[1]}: = m2_v1/summary.csv ({src})", f"{st['pairs_ttbar']:.3f} / {st['k'] / st['n']:.6f}",
              f"{r.pairs_ttbar:.3f} / {r.ceil_primary:.6f}", ok)
    m2key = {v: k for k, v in M2_ROWS.items()}
    notes = []
    for lab, s, field, q, f, ln in REFS:

        if (lab, s, field) in SAMEPID:

            st = stats[(lab, s)]
            notes.append(f"{f}:{ln} ({lab}, seed {s}): quoted {q}; dm_ceiling {st['k'] / st['n']:.4f} "
                         f"({st['k']} of {st['n']}); same-particle ceiling {sp:.4f}")
            continue
        check(f"{lab} s{s}: {field} as quoted on {f}:{ln}", got, q, got == q and q in line)

    t = pd.DataFrame(rows)
    bad = [c for c in CHECKS if not c[3]]
    write(a.out, t, stats, notes)
    print(f"{len(CHECKS)} checks, {len(bad)} failed")
    for c in bad:
        print("FAILED:", c)
    return 1 if bad else 0


def write(out: str, t: pd.DataFrame, stats: dict, notes: list[str]) -> None:
    job = os.environ.get("SLURM_JOB_ID", "interactive")
    head = subprocess.run(["git", "rev-parse", "--short", "HEAD"], capture_output=True, text=True).stdout.strip()
    now = datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%d %H:%M UTC")
    L = ["# Model table (paper draft 7, Table II): facts read from the stored run files (validation)", "",
         f"Generated by the script at the end of this file (Slurm job {job}, {now}, repo HEAD {head}). Validation",
         "events only (2999: 1000 ttbar, 1000 hnl10_100, 999 hnl15_200); no test file is read and nothing is chosen.",
         "Pairs per ttbar event: graph edges (unique hit pairs, eval_val/per_event.parquet n_edges), mean over the 1000",

         "## Quoted values (seed 0)", ""]
    for label, *_ in MODELS:

                 f"({st['pairs_ttbar']:,.3f} over {st['n_ttbar_events']} events)")
        L.append(f"- {label}, seed 0: primary graph-stage ceiling {100 * st['k'] / st['n']:.2f} % "
                 f"({st['k']} of {st['n']})")
    for label in ("DeZoort IN", "DeZoort IN-L"):

        L.append(f"- {label}: one seed (0), {r.train_graphs} ttbar training graphs, {r.epochs} epochs, {r.steps:,} steps, "
